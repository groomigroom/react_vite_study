.filter

리액트 TSX에서 .filter 메서드는 주로 
인터페이스(Interface)나 타입(Type)이 지정된 
배열 데이터를 조건에 따라 골라내어 화면에 
렌더링하거나, 특정 아이템을 삭제(제외)할 때 
사용합니다.tsx

import React, { useState } from 'react';

interface User {
  id: number;
  name: string;
  isAdmin: boolean;
}

export default function UserList() {
  const [users, setUsers] = useState<User[]>([
    { id: 1, name: '철수', isAdmin: false },
    { id: 2, name: '영희', isAdmin: true },
    { id: 3, name: '민수', isAdmin: false },
  ]);

  // 관리자만 골라내는 filter 예제
  const adminUsers = users.filter((user) => user.isAdmin);

  // 특정 id를 가진 유저를 삭제(제외)하는 예제
  const removeUser = (id: number) => {
    setUsers(users.filter((user) => user.id !== id));
  };

  return (
    <div>
      <h3>관리자 목록</h3>
      <ul>
        {adminUsers.map((user) => (
          <li key={user.id}>{user.name}</li>
        ))}
      </ul>
    </div>
  );
}


주요 활용 방법
조건부 렌더링: .filter()로 특정 
조건(예: user.isAdmin)에 맞는 
요소만 남긴 뒤 .map()으로 화면에 출력합니다.
데이터 삭제: 불변성을 지키며 삭제하려는 
아이템의 id가 아닌 항목들만 남겨 새로운 배열을 만듭니다.
검색 기능: 입력창(input)의 검색어 
상태값과 일치하는 항목만 필터링합니다.
