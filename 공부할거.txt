React TSX(TypeScript)에서 useRef는 리렌더링을 유발하지 않고 값을 유지하거나 특정 DOM 요소에 직접 접근할 때 사용하는 훅입니다. TSX 환경에서는 안전한 타입 지정을 위해 제네릭을 활용합니다.주요 의미와 용도DOM 직접 제어: 입력창 포커스, 스크롤 위치 이동 등 HTML 요소에 직접 접근.값 저장: 타이머 ID, 이전 값 등 화면 표출(렌더링)과 관계없는 데이터를 보존.리렌더링 방지: current 값이 바뀌어도 컴포넌트가 다시 그려지지 않아 성능 최적화에 기여.TypeScript(TSX)에서의 특징타입 추론과 제네릭: useRef<HTMLInputElement>(null)처럼 초기값과 DOM 타입을 명시해야 안전합니다.Readonly vs Mutable: 초기값 유무에 따라 읽기 전용(RefObject) 또는 수정 가능(MutableRefObject) 타입으로 다르게 지정됩니다.

import { useState } from 'react';

interface Todo {
  id: number;
  text: string;
}

export default function App() {
  const [todos, setTodos] = useState<Todo[]>([
    { id: 1, text: "React 공부하기" },
    { id: 2, text: "맛있는 저녁 먹기" },
  ]);

  const deleteTodo = (id: number) => {
    setTodos(todos.filter(todo => todo.id !== id));
  };

  return (
    <ul>
      {todos.map(todo => (
        <li key={todo.id}>
          {todo.text}
          <button onClick={() => deleteTodo(todo.id)}>삭제</button>
        </li>
      ))}
    </ul>
  );
}

  );

    <div>
      <input value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)} />
      <ul>
        {/* 필터링된 배열만 화면에 출력 */}
        {filteredUsers.map(user => <li key={user}>{user}</li>)}
      </ul>
    </div>
  );
}

import { useState } from 'react';

interface Product {
  name: string;
  category: 'electronics' | 'clothing';
}

const products: Product[] = [
  { name: '노트북', category: 'electronics' },
  { name: '셔츠', category: 'clothing' },
  { name: '이어폰', category: 'electronics' },
  { name: '바지', category: 'clothing' },
];

export default function App() {
  const [selectedCategory, setSelectedCategory] = useState<'all' | 'electronics' | 'clothing'>('all');

  const filteredProducts = selectedCategory === 'all'
    ? products
    : products.filter(product => product.category === selectedCategory);

  return (
    <div style={{ padding: '20px' }}>
      <h2>상품 목록</h2>

      <div style={{ marginBottom: '15px' }}>
        <button onClick={() => setSelectedCategory('all')}>전체보기</button>
        <button onClick={() => setSelectedCategory('electronics')} style={{ marginLeft: '10px' }}>전자기기</button>
        <button onClick={() => setSelectedCategory('clothing')} style={{ marginLeft: '10px' }}>의류</button>
      </div>

      <ul>
        {filteredProducts.map((product, index) => (
          <li key={index}>
            {product.name} ({product.category})
          </li>
        ))}
      </ul>
    </div>
  );
}

import { useEffect } from 'react';

const EnglishPage = () => {
    useEffect(() => {
        // 페이지 진입 시 영어로 설정
        document.documentElement.lang = 'en';

        // 페이지를 나갈 때 기존 언어(ko)로 복구
        return () => {
            document.documentElement.lang = 'ko';
        };
    }, []);

    return (
        <div>
            <h1>Welcome to the English Page</h1>
            <p>This content is optimized for English speakers.</p>
        </div>
    );
};

export default EnglishPage;
